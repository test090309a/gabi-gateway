Um das aktuelle Wetter in Wien zu überprüfen, könntest du eine API wie OpenWeatherMap nutzen. Da ich keine Echtzeitdaten abrufen kann, werde ich dir aber einen Beispielcode in Python zeigen, wie du das machen könntest:

```python
import requests

def check_weather(city, api_key):
    base_url = "http://api.openweathermap.org/data/2.5/weather?"
    complete_url = base_url + f"q={city}&appid={api_key}&units=metric"  # Metriken (Grad Celsius)

    response = requests.get(complete_url)
    data = response.json()

    if response.status_code == 200:
        main = data['main']
        temp = main['temp']
        weather_description = data['weather'][0]['description']
        
        weather_info = f"Temperatur: {temp}°C\nWetterbeschreibung: {weather_description}"
        return weather_info
    else:
        return "Fehler beim Abrufen der Wetterdaten"

# Beispielaufruf
api_key = "YOUR_OPENWEATHERMAP_API_KEY"  # Hier deinen API-Schlüssel einfügen
city = "Wien"
weather_info = check_weather(city, api_key)
print(weather_info)
```
